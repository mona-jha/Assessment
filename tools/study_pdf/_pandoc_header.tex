% Suppress pandoc's default \maketitle hook contents, install our cover.
% Pandoc sets \@title / \@author / \@date via \title{...} etc.; we need
% \makeatletter so '@' is treated as a letter inside the macro body.
\makeatletter
\renewcommand{\maketitle}{%
  \begin{titlepage}
    \newgeometry{margin=0pt}
    \begin{tikzpicture}[remember picture, overlay]
      % Dark navy band (top ~32%)
      \fill[covernavy]
        (current page.south west) rectangle (current page.north east);
      % Grey-blue band (lower ~68%)
      \fill[covergrey]
        ([yshift=-0.32\paperheight]current page.north west) rectangle
        (current page.south east);
      % Hairline rule at zone boundary
      \draw[white, opacity=0.12, line width=0.4pt]
        ([yshift=-0.32\paperheight]current page.north west) --
        ([yshift=-0.32\paperheight]current page.north east);
      % Golden vertical rule in navy zone
      \draw[springergold, opacity=0.7, line width=1.2pt]
        ([xshift=1.8cm, yshift=-0.32\paperheight]current page.north west) --
        ([xshift=1.8cm]current page.north west);
      % Faint vertical line in grey zone
      \draw[covergrey!35!black, opacity=0.5, line width=0.8pt]
        ([xshift=1.8cm]current page.south west) --
        ([xshift=1.8cm, yshift=-0.32\paperheight]current page.north west);
      % Subtle right-edge shading (book-spine effect)
      \shade[left color=covergrey, right color=covergrey!94!white]
        ([xshift=-0.5\paperwidth, yshift=-0.32\paperheight]current page.north east)
        rectangle (current page.south east);
      \shade[left color=covernavy, right color=covernavy!94!white]
        ([xshift=-0.5\paperwidth]current page.north east) rectangle
        ([yshift=-0.32\paperheight]current page.north east);
      %
      % Series label (top-left, in dark zone)
      \node[anchor=north west, text width=0.6\paperwidth]
        at ([xshift=2.2cm, yshift=-1.2cm]current page.north west)
        {\color{white!85}\fontsize{11}{14}\selectfont\sffamily
         Personal Study Notes};
      % Author (gold, just above the rule)
      \node[anchor=south west, text width=0.75\paperwidth]
        at ([xshift=2.2cm, yshift=-0.29\paperheight]current page.north west)
        {\color{springergold}\fontsize{16}{20}\selectfont\sffamily \@author};
      % Gold accent block descending from the rule
      \fill[springergold]
        ([yshift=-0.32\paperheight]current page.north west) rectangle
        ([xshift=1.8cm, yshift=-0.49\paperheight]current page.north west);
      % Module badge
      \node[anchor=north west, text width=0.75\paperwidth]
        at ([xshift=2.2cm, yshift=-0.345\paperheight]current page.north west)
        {\color{white!70}\fontsize{15}{18}\selectfont\sffamily
         Syngene / BBRC Digital Pathology Assessment};
      % Title (large, light, white)
      \node[anchor=north west, text width=0.82\paperwidth]
        at ([xshift=2.2cm, yshift=-0.39\paperheight]current page.north west)
        {\color{white}\fontsize{40}{52}\selectfont\fontseries{l}\selectfont\sffamily
         \hyphenpenalty=10000\exhyphenpenalty=10000\relax \@title\par};
      % Subtitle (gold)
      \node[anchor=north west, text width=0.75\paperwidth]
        at ([xshift=2.2cm, yshift=-0.55\paperheight]current page.north west)
        {\color{springergold}\fontsize{19}{25}\selectfont\sffamily
         Self-Study Companion\par};
      % Date (bottom-right)
      \node[anchor=south east]
        at ([xshift=-1.5cm, yshift=1.5cm]current page.south east)
        {\color{white!55}\fontsize{11}{14}\selectfont\sffamily \@date};
    \end{tikzpicture}
    \restoregeometry
  \end{titlepage}
  % Reset \vsize after \restoregeometry so TOC paginates correctly.
  \global\vsize=\textheight
  \pagegoal=\textheight
}
